```python
def compute_PrimaryAuthsystem_nodes(
    self,
    remaining_accounts_list,      # only delta/changed accounts from EDW
    primary_account_domains,
    neo4j_primary=None,           # current Primary in Neo4j: {"domain": ..., "samaccountname": ...}
    neo4j_all_user_accounts=None  # ALL 4 accounts currently in Neo4j for this user (required!)
):
    """
    Handles delta input + full PrimaryAuth switch with "same login ID" rule
    Returns: full list of ALL accounts to update in Neo4j (usually 4)
    """
    try:
        data = remaining_accounts_list[0]
        delta_accounts = data["user_accounts"]  # only changed ones
        oneid = data["oneID"]
        emp_id = data["E_EmployeeID"]

        # STEP 1: Detect NEW Primary domain from the changed data
        new_primary_domain = None
        for acc in delta_accounts:
            if (acc["domain"] == "CORP.CVSCAREMARK" and
                str(acc.get("extensionattribute3", "")).lower() == "cloud"):
                new_primary_domain = "CORP.CVSCAREMARK"
                break

        # If no CLOUD flag appeared → no Primary change → return only delta
        if not new_primary_domain:
            # Just update the changed accounts, no Primary switch
            for acc in delta_accounts:
                acc["oneID"] = oneid
                acc["CVSResourceid"] = emp_id
            return delta_accounts, primary_account_domains

        # STEP 2: Get old Primary from Neo4j
        old_primary_sam = neo4j_primary.get("samaccountname") if neo4j_primary else None
        old_primary_domain = neo4j_primary.get("domain") if neo4j_primary else None

        if not old_primary_sam:
            # First time ever
            for acc in delta_accounts:
                acc["PrimaryAuthsystem"] = (acc["domain"] == new_primary_domain)
                acc["AccountType"] = "Primary" if acc["PrimaryAuthsystem"] else "Secondary"
                acc["oneID"] = oneid
                acc["CVSResourceid"] = emp_id
            primary_account_domains[1] = [new_primary_domain]
            return delta_accounts, primary_account_domains

        # STEP 3: PRIMARY DOMAIN CHANGED → Apply "same login ID" rule
        if old_primary_domain != new_primary_domain:
            print(f"PRIMARY SWITCH: {old_primary_sam}@{old_primary_domain} → {old_primary_sam}@{new_primary_domain}")

            # We must return ALL accounts (not just delta)
            all_accounts_to_update = []

            # Use neo4j_all_user_accounts as base (has all 4)
            for neo_acc in neo4j_all_user_accounts:
                acc = neo_acc.copy()
                acc["oneID"] = oneid
                acc["CVSResourceid"] = emp_id

                # Same samaccountname in NEW domain → Primary
                if (acc["samaccountname"] == old_primary_sam and
                    acc["domain"] == new_primary_domain):
                    acc["PrimaryAuthsystem"] = True
                    acc["AccountType"] = "Primary"
                else:
                    acc["PrimaryAuthsystem"] = False
                    acc["AccountType"] = "Secondary"

                all_accounts_to_update.append(acc)

            # Update tracking
            primary_account_domains[1] = [new_primary_domain]
            primary_account_domains[2].append(old_primary_sam)

            return all_accounts_to_update, primary_account_domains

        # No domain switch → just return delta
        for acc in delta_accounts:
            acc["oneID"] = oneid
            acc["CVSResourceid"] = emp_id
        return delta_accounts, primary_account_domains

    except Exception as e:
        logging.error(f"Error in compute_PrimaryAuthsystem_nodes: {e}")
        return delta_accounts, primary_account_domains
```
