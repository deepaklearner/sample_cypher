Grok:

UNWIND $rows AS row
MATCH (usr:User {employeeNumber: row.CVSResourceid})-[:HAS_ATTRIBUTE]->(usrAct:UserAccount)

// First phase: remove HAS_ACCOUNT_MAPPING relationships pointing to this account from Ping Directory
OPTIONAL MATCH (pingAccount:UserAccount {targetSystem: "Ping Directory"})
    -[mapping:HAS_ACCOUNT_MAPPING]->(usrAct)
DELETE mapping

// Second phase: delete all non-primary UserAccount nodes (except the primary Ping one)
WHERE NOT (usrAct.targetSystem = 'Ping Directory' AND usrAct.accountType = 'Primary')

DETACH DELETE usrAct

RETURN COUNT(DISTINCT usr) AS totalProcessed

UNWIND $rows AS row
MATCH (usr:User {employeeNumber: row.CVSResourceid})

WITH usr, row
MATCH (usr)-[:HAS_ATTRIBUTE]->(usrAct:UserAccount)

// Clean incoming mappings from Ping Directory accounts
OPTIONAL MATCH (ping:UserAccount {targetSystem: "Ping Directory"})
    -[mapping:HAS_ACCOUNT_MAPPING]->(usrAct)
DELETE mapping

// Now delete all non-primary accounts
WITH usr, usrAct
WHERE NOT (usrAct.targetSystem = 'Ping Directory' AND usrAct.accountType = 'Primary')
DETACH DELETE usrAct

RETURN COUNT(DISTINCT usr) AS total
