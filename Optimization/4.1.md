how to find owner_added and owner_removed?

// Assume $entitlementName, $targetSystem, $edwOwners is a parameter list of owner ids
MATCH (e:Entitlement {name: $entitlementName, targetSystem: $targetSystem})
OPTIONAL MATCH (e)-[:HAS_OWNER]->(u:User)
WITH e, collect(u.employeeNumber) + collect(u.aetnaresourceid) + collect(u.cvsresourceid) AS currentOwners, $edwOwners AS mysqlOwners
WITH e, currentOwners, mysqlOwners,
     [owner IN mysqlOwners WHERE NOT owner IN currentOwners] AS owner_to_add,
     [owner IN currentOwners WHERE NOT owner IN mysqlOwners] AS owner_to_remove
RETURN e, owner_to_add, owner_to_remove



Remove old owners:

UNWIND $owner_removed AS row
MATCH (e:Entitlement {entitlementName: row.entitlementName, targetSystem: row.targetSystem})
MATCH (e)-[r:HAS_OWNER]->(u:User)
WHERE u.employeeNumber = row.ownerId 
   OR u.aetnaresourceid = row.ownerId
   OR u.cvsresourceid = row.ownerId
DELETE r

Add new owners:

UNWIND $owner_added AS row
MATCH (e:Entitlement {entitlementName: row.entitlementName, targetSystem: row.targetSystem})
MATCH (u:User)
WHERE u.employeeNumber = row.ownerId 
   OR u.aetnaresourceid = row.ownerId
   OR u.cvsresourceid = row.ownerId
MERGE (e)-[:HAS_OWNER]->(u)
