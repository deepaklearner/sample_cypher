```python
def compute_PrimaryAuthsystem_nodes(
    self,
    remaining_accounts_list,
    primary_account_domains,
    neo4j_primary=None,                # ← NEW: current Primary from Neo4j
    neo4j_all_user_accounts=None       # ← NEW: all accounts from Neo4j (required for delta)
):
    try:
        return_var = None
        ordered_accounts = remaining_accounts_list.copy()
        accounts = ordered_accounts["user_accounts"]

        # Keep your original AETH-first ordering
        aeth_accounts = [a for a in accounts if a.get("domain") == "AETH"]
        other_accounts = [a for a in accounts if a.get("domain") != "AETH"]
        ordered_user_accounts = aeth_accounts + other_accounts
        ordered_accounts["user_accounts"] = ordered_user_accounts

        # === STEP 1: Find NEW PrimaryAuth domain from current EDW data ===
        primary_auth_domains = []
        for acc in ordered_user_accounts:
            if (
                (acc["domain"] == "AETH" and
                 acc["samaccountname"].lower().startswith(("a", "n")) and
                 str(acc.get("admin_description", "")) in ["nan", "NULL", "", "DNE", None])
                or
                (acc["domain"].lower() in self.default_ad_list and
                 "cloud" in str(acc.get("extensionattribute", "")).lower())
            ):
                primary_auth_domains.append(acc["domain"])

        # Take first one as authoritative domain
        new_primary_auth_domain = primary_auth_domains[0] if primary_auth_domains else None

        # === STEP 2: If no change or first time → use old logic ===
        if not new_primary_auth_domain:
            # Your original fallback logic
            for acc in ordered_user_accounts:
                if len(primary_auth_domains) == 1 and acc["domain"] == primary_auth_domains[0]:
                    acc["PrimaryAuthsystem"] = True
                    acc["AccountType"] = "Primary"
                    acc["oneID"] = ordered_accounts["oneID"]
                    acc["CVSResourceid"] = ordered_accounts["E_EmployeeID"]
                    return_var = acc
                    primary_account_domains[1] = [acc["domain"]]
                    primary_account_domains[2].append(acc["samaccountname"])
                    break
            return return_var, primary_account_domains

        # === STEP 3: PRIMARY CHANGE → Apply AETNA RULE ===
        # Find which samaccountname should be the Primary login ID
        primary_sam = None
        for acc in ordered_user_accounts:
            if acc["domain"] == new_primary_auth_domain:
                primary_sam = acc["samaccountname"]
                break

        if not primary_sam and neo4j_primary:
            primary_sam = neo4j_primary.get("samaccountname")  # fallback

        # If still no sam → use first account
        if not primary_sam and ordered_user_accounts:
            primary_sam = ordered_user_accounts[0]["samaccountname"]

        # === FINAL STEP: Update ALL accounts (from Neo4j) with Aetna rule ===
        updated_accounts = []
        for acc in (neo4j_all_user_accounts or ordered_user_accounts):
            acc = dict(acc)  # copy
            acc["oneID"] = ordered_accounts["oneID"]
            acc["CVSResourceid"] = ordered_accounts["E_EmployeeID"]

            if acc["samaccountname"] == primary_sam:
                acc["AccountType"] = "Primary"
                acc["PrimaryAuthsystem"] = (acc["domain"] == new_primary_auth_domain)
            else:
                acc["AccountType"] = "Secondary"
                acc["PrimaryAuthsystem"] = False

            updated_accounts.append(acc)

            # Set return_var to the one with PrimaryAuthsystem = True
            if acc["PrimaryAuthsystem"]:
                return_var = acc

        # Update tracking
        primary_account_domains[0].append(new_primary_auth_domain)
        primary_account_domains[1] = [new_primary_auth_domain]
        primary_account_domains[2].append(primary_sam)

        print(f"AETNA RULE: {primary_sam} is Primary login | Auth domain: {new_primary_auth_domain}")

        return updated_accounts, primary_account_domains  # ← Now returns ALL updated accounts

    except Exception as e:
        logging.error(f"Exception in compute_PrimaryAuthsystem_nodes: {str(e)}")
        return return_var, primary_account_domains
```

| Detects new PrimaryAuth domain from EDW | Same logic you had |
| Final loop applies **Aetna rule** across **all accounts** | This is the only real change |
