def compute_PrimaryAuthsystem_nodes(self, remaining_accounts_list, primary_secondary_account_domains):
    try:
        return_var = None
        ordered_accounts = remaining_accounts_list.copy()
        accounts = ordered_accounts["user_accounts"]

        # Keep your original AETH-first ordering
        aeth_accounts = [a for a in accounts if a.get("domain") == "AETH"]
        other_accounts = [a for a in accounts if a.get("domain") != "AETH"]
        ordered_user_accounts = aeth_accounts + other_accounts
        ordered_accounts["user_accounts"] = ordered_user_accounts

        # STEP 1: Detect PrimaryAuth domain from EDW (your original logic)
        candidate_domains = []
        for acc in ordered_user_accounts:
            if (
                (acc["domain"] == "AETH" and
                 acc["samaccountname"].lower().startswith(("a", "n")) and
                 str(acc.get("admin_description", "")) in ["nan", "NULL", "", "DNE", None])
                or
                (acc["domain"].lower() in self.default_ad_list and
                 "cloud" in str(acc.get("extensionattribute", "")).lower())
            ):
                candidate_domains.append(acc["domain"])

        new_primary_auth_domain = candidate_domains[0] if candidate_domains else None

        # STEP 2: FIRST-TIME INSERT FIX → If no PrimaryAuth domain exists yet
        if not primary_secondary_account_domains[1] and new_primary_auth_domain:
            # This is the FIRST time we are setting PrimaryAuth
            primary_login_sam = None
            for acc in ordered_user_accounts:
                if acc["domain"] == new_primary_auth_domain:
                    primary_login_sam = acc["samaccountname"]
                    acc["PrimaryAuthsystem"] = True
                    acc["AccountType"] = "Primary"
                    acc["oneID"] = ordered_accounts["oneID"]
                    acc["CVSResourceid"] = ordered_accounts["E_EmployeeID"]
                    return_var = acc
                    break

            # Set all OTHER accounts as Primary (AccountType) or Secondary
            all_accounts = ordered_user_accounts.copy()
            for acc in all_accounts:
                acc["oneID"] = ordered_accounts["oneID"]
                acc["CVSResourceid"] = ordered_accounts["E_EmployeeID"]
                if acc["samaccountname"] == primary_login_sam:
                    acc["AccountType"] = "Primary"
                    acc["PrimaryAuthsystem"] = (acc["domain"] == new_primary_auth_domain)
                else:
                    acc["AccountType"] = "Secondary"
                    acc["PrimaryAuthsystem"] = False

            # Update tracking
            primary_secondary_account_domains[0].append(new_primary_auth_domain)
            primary_secondary_account_domains[1] = [new_primary_auth_domain]
            primary_secondary_account_domains[2].append(primary_login_sam)

            # Rebuild primary/secondary lists
            primary_secondary_account_domains[4] = [a for a in all_accounts if a["AccountType"] == "Primary" and a["PrimaryAuthsystem"]]
            primary_secondary_account_domains[5] = [a for a in all_accounts if a["AccountType"] == "Secondary" or not a["PrimaryAuthsystem"]]

            print(f"FIRST-TIME INSERT: {primary_login_sam}@{new_primary_auth_domain} set as PrimaryAuth")
            return all_accounts, primary_secondary_account_domains

        # STEP 3: NORMAL CASE (already exists in Neo4j)
        if new_primary_auth_domain:
            primary_login_sam = None
            for acc in ordered_user_accounts:
                if acc["domain"] == new_primary_auth_domain:
                    primary_login_sam = acc["samaccountname"]
                    break

            if not primary_login_sam and primary_secondary_account_domains[4]:
                primary_login_sam = primary_secondary_account_domains[4][0]["samaccountname"]

            # Use ALL accounts from index 4 + 5
            all_accounts = (primary_secondary_account_domains[4] + primary_secondary_account_domains[5])
            updated_accounts = []

            for acc in all_accounts:
                acc = dict(acc)
                acc["oneID"] = ordered_accounts["oneID"]
                acc["CVSResourceid"] = ordered_accounts["E_EmployeeID"]

                if acc["samaccountname"] == primary_login_sam:
                    acc["AccountType"] = "Primary"
                    acc["PrimaryAuthsystem"] = (acc["domain"] == new_primary_auth_domain)
                else:
                    acc["AccountType"] = "Secondary"
                    acc["PrimaryAuthsystem"] = False

                updated_accounts.append(acc)
                if acc["PrimaryAuthsystem"]:
                    return_var = acc

            primary_secondary_account_domains[0].append(new_primary_auth_domain)
            primary_secondary_account_domains[1] = [new_primary_auth_domain]
            primary_secondary_account_domains[2].append(primary_login_sam)
            primary_secondary_account_domains[4] = [a for a in updated_accounts if a["AccountType"] == "Primary" and a["PrimaryAuthsystem"]]
            primary_secondary_account_domains[5] = [a for a in updated_accounts if a["AccountType"] == "Secondary" or not a["PrimaryAuthsystem"]]

            return updated_accounts, primary_secondary_account_domains

        # STEP 4: No Primary change → old behavior
        for acc in ordered_user_accounts:
            if (len(candidate_domains) == 1 and 
                acc["domain"] == candidate_domains[0]):
                acc["PrimaryAuthsystem"] = True
                acc["AccountType"] = "Primary"
                acc["oneID"] = ordered_accounts["oneID"]
                acc["CVSResourceid"] = ordered_accounts["E_EmployeeID"]
                return_var = acc
                primary_secondary_account_domains[1] = [acc["domain"]]
                primary_secondary_account_domains[2].append(acc["samaccountname"])
                break

        return return_var, primary_secondary_account_domains

    except Exception as e:
        logging.error(f"Exception: {str(e)}")
        return return_var, primary_secondary_account_domains
