```python
def compute_deltas(current_edw, neo4j_data):
    results = []

    all_keys = set(current_edw.keys()) | set(neo4j_data.keys())

    for key in all_keys:
        edw_owners = set(current_edw.get(key, []))
        neo4j_owners_info = neo4j_data.get(key, [])

        active_neo4j = set(u['employeeNumber'] for u in neo4j_owners_info if u['isActive'])
        inactive_neo4j = set(u['employeeNumber'] for u in neo4j_owners_info if not u['isActive'])
        all_neo4j = active_neo4j | inactive_neo4j

        missing   = edw_owners - all_neo4j
        inactive  = edw_owners & inactive_neo4j
        matched   = edw_owners & active_neo4j
        removed   = all_neo4j - edw_owners  # ✅ updated definition

        results.append({
            "entitlementName": key[0],
            "targetSystem": key[1],
            "missing": list(missing),
            "inactive": list(inactive),
            "matched": list(matched),
            "removed": list(removed)
        })

    return results
```
