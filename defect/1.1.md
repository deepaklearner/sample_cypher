UNWIND $rows AS row

// Find the User once
MATCH (usr:User {employeeNumber: row.CVSResourceid})

// Collect all UserAccount nodes that should be deleted
WITH usr, row
MATCH (usr)-[r:HAS_ATTRIBUTE]->(usrAct:UserAccount)
WHERE usrAct.accountType IN ['Primary', 'Secondary']
  AND NOT (usrAct.targetSystem = 'Ping Directory' AND usrAct.accountType = 'Primary')

// Delete each one only once, even if the same user appears multiple times in $rows
WITH DISTINCT usrAct
DETACH DELETE usrAct

RETURN COUNT(*) AS total
