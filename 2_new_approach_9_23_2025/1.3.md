### 🧩 1. `main.py` — The Orchestrator

```python
from extract.primary_extractor import get_primary_domains, fetch_data as fetch_primary_data
from extract.non_primary_extractor import get_non_primary_domains, fetch_batch as fetch_non_primary_batch
from transform.transformer import transform_primary, transform_non_primary
from load.neo4j_loader import load_to_neo4j
from utils.helpers import chunk
from config.settings import BATCH_SIZE_NON_PRIMARY

def main():
    # Step 1: Primary Domains
    primary_domains = get_primary_domains()
    for domain in primary_domains:
        data = fetch_primary_data(domain)
        transformed = transform_primary(data)
        load_to_neo4j(transformed)

    # Step 2: Non-Primary Domains
    non_primary_domains = get_non_primary_domains()
    for batch in chunk(non_primary_domains, BATCH_SIZE_NON_PRIMARY):
        data = fetch_non_primary_batch(batch)
        transformed = transform_non_primary(data)
        load_to_neo4j(transformed)

if __name__ == "__main__":
    main()
```

    for i in range(0, len(lst), size):
        yield lst[i:i + size]

import logging

def get_logger(name):
    logger = logging.getLogger(name)
    if not logger.handlers:
        handler = logging.StreamHandler()
        formatter = logging.Formatter('%(asctime)s [%(levelname)s] %(message)s')
        handler.setFormatter(formatter)
        logger.addHandler(handler)
        logger.setLevel(logging.INFO)
    return logger
