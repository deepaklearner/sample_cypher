def delta_src_vs_gdb_usrs_usraccts_profiles(
    self, src_usraccts_for_each_usr: pd.DataFrame, gdb_usrs_with_usraccts_has_oneid_list: pd.DataFrame
):
    """
    Detect delta users for processing:
        - New accounts (concat_attr_col)
        - Updated accounts (concat_attr_col1)
        - PAS-changed users (NEW REQUIREMENT)
        - Removed accounts
    """
    logging.info("Identifying delta of users and useraccounts (PAS-aware)")

    new_pas_data = []
    removed_user_accounts = []

    df1 = pd.DataFrame(src_usraccts_for_each_usr)      # EDW data
    df2 = pd.DataFrame(gdb_usrs_with_usraccts_has_oneid_list)  # Neo4j data

    if df1.empty:
        df1['concat_attr_col2'] = None

    # --------------------------------------------------------------------
    # STEP 1: Detect Removed Accounts
    # --------------------------------------------------------------------
    removed_accounts = set(df2['concat_attr_col2']) - set(df1['concat_attr_col2']) \
        if df2['concat_attr_col2'].tolist() != [None] else set()

    for account in removed_accounts:
        if account is not None and 'UNASSIGNED' not in account:
            emp_id, sam, dom, *_ = account.split('|')
            removed_user_accounts.append({
                'CVSResourceid': emp_id,
                'samaccountname_r': sam,
                'domain_n': dom,
                'label': 'UNASSIGNED'
            })

    logging.info(f"Total removed user accounts: {len(removed_user_accounts)}")

    # --------------------------------------------------------------------
    # STEP 2: Identify New and Updated Accounts
    # --------------------------------------------------------------------
    df1['usracct_oneid'] = None

    # New accounts (identity-level)
    diff_accounts = set(df1['concat_attr_col']) - set(df2['concat_attr_col'])

    # Updated accounts (attribute-level)
    updated_accounts = set(df1['concat_attr_col1']) - set(df2['concat_attr_col1'])

    # IMPORTANT:
    # updated_df removes accounts where concat_attr_col also changed
    # meaning: account exists but properties have changed
    updated_df = df1.loc[
        df1["concat_attr_col1"].isin(updated_accounts) &
        ~df1["concat_attr_col"].isin(diff_accounts)
    ]

    updated_users = set(updated_df["E_EmployeeID"])

    # --------------------------------------------------------------------
    # STEP 3: PAS CHANGE DETECTION (NEW REQUIREMENT)
    # --------------------------------------------------------------------
    pas_change_users = set()

    # Neo4j known primaryAuth accounts
    neo4j_primary = (
        df2[df2["PrimaryAuth"] == True]
        .groupby("E_EmployeeID")["targetSystem"]
        .first()
        .to_dict()
    )

    # PAS RULE: EDW account is primaryAuth if:
    # - targetSystem in CORP, CSARMARK, CVS
    # - extensionattribute3 contains keyword: cloud
    def compute_edw_pas(row):
        if (
            row['domain'] in ['CORP', 'CSARMARK', 'CVS'] or
            row['domain'].upper().startswith("CORP") or
            "CVS" in row['domain']
        ):
            ext = str(row.get("extensionattribute", "")).lower()
            if "cloud" in ext:
                return True
        return False

    df1["EDW_PrimaryAuth"] = df1.apply(compute_edw_pas, axis=1)

    # EDW PAS lookup
    edw_primary = (
        df1[df1["EDW_PrimaryAuth"] == True]
        .groupby("E_EmployeeID")["domain"]
        .first()
        .to_dict()
    )

    # Detect PAS change
    for emp in set(df1["E_EmployeeID"]):
        edw_pas = edw_primary.get(emp)
        neo_pas = neo4j_primary.get(emp)

        if edw_pas != neo_pas:
            pas_change_users.add(emp)

    logging.info(f"Users with PAS change: {len(pas_change_users)}")

    # --------------------------------------------------------------------
    # STEP 4: Build Final Delta User List
    # --------------------------------------------------------------------

    # New accounts users
    new_users = set(df1.loc[df1["concat_attr_col"].isin(diff_accounts)]["E_EmployeeID"])

    # Aggregate final users needing processing
    relevant_employees = new_users | updated_users | pas_change_users

    logging.info(f"Total users needing delta processing: {len(relevant_employees)}")

    # Build df1 subset
    df1_tbm = df1[df1["E_EmployeeID"].isin(relevant_employees)][[
        "E_EmployeeID", "usracct_oneid", "samaccountname", "domain",
        "extensionattribute", "admin_description"
    ]].copy()

    # Merge with user_oneid from Neo4j
    df2_master = df2[['E_EmployeeID', 'user_oneid', 'lastuseraccountoneid']].drop_duplicates(
        subset='E_EmployeeID', keep='first'
    )

    df1_tbm = pd.merge(df1_tbm, df2_master, on='E_EmployeeID', how='left')

    # --------------------------------------------------------------------
    # STEP 5: Build structured grouped data
    # --------------------------------------------------------------------
    try:
        grouped_users = (
            df1_tbm.groupby('E_EmployeeID')
            .apply(lambda g: self.extract(g))
            .values
        )
        users_details_list1 = pd.DataFrame(list(grouped_users))
    except Exception as e:
        logging.error(f"Error building user details list: {e}")
        users_details_list1 = pd.DataFrame()

    # --------------------------------------------------------------------
    # Return results
    # --------------------------------------------------------------------
    return users_details_list1, removed_user_accounts, new_pas_data
