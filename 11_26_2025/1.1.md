def compute_PrimaryAuthsystem_nodes(
    self,
    remaining_accounts_list,
    primary_account_domains,
    neo4j_primary=None,
    neo4j_all_user_accounts=None
):
    try:
        data = remaining_accounts_list[0]
        delta_accounts = data["user_accounts"]
        oneid = data["oneID"]
        emp_id = data["E_EmployeeID"]

        # STEP 1: Detect NEW Primary domain from EDW delta
        new_primary_domain = None
        for acc in delta_accounts:
            if (acc["domain"] == "CORP.CVSCAREMARK" and
                str(acc.get("extensionattribute3", "")).lower() == "cloud"):
                new_primary_domain = "CORP.CVSCAREMARK"
                break

        if not new_primary_domain:
            # No Primary change → just return delta
            for acc in delta_accounts:
                acc["oneID"] = oneid
                acc["CVSResourceid"] = emp_id
            return delta_accounts, primary_account_domains

        # STEP 2: Get old Primary samaccountname
        old_primary_sam = neo4j_primary.get("samaccountname") if neo4j_primary else None
        old_primary_domain = neo4j_primary.get("domain") if neo4j_primary else None

        if not old_primary_sam:
            old_primary_sam = delta_accounts[0]["samaccountname"]  # fallback

        # STEP 3: Apply "Same Login ID" rule
        # → old_primary_sam loses Primary
        # → old_primary_sam in NEW domain becomes Primary
        all_accounts_to_update = []

        for acc in neo4j_all_user_accounts:
            updated = acc.copy()
            updated["oneID"] = oneid
            updated["CVSResourceid"] = emp_id

            # Core Rule:
            if (updated["samaccountname"] == old_primary_sam and
                updated["domain"] == new_primary_domain):
                updated["PrimaryAuthsystem"] = True
                updated["AccountType"] = "Primary"
            else:
                updated["PrimaryAuthsystem"] = False
                updated["AccountType"] = "Secondary"

            all_accounts_to_update.append(updated)

        # Update tracking
        primary_account_domains[1] = [new_primary_domain]
        primary_account_domains[2].append(old_primary_sam)

        print(f"PRIMARY SWITCH: {old_primary_sam}@{old_primary_domain} → {old_primary_sam}@{new_primary_domain}")

        return all_accounts_to_update, primary_account_domains

    except Exception as e:
        logging.error(f"Error: {e}")
        return delta_accounts, primary_account_domains
